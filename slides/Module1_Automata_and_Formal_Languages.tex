\begin{frame}{Week 1: Core Mathematical Foundations}
  \begin{tcolorbox}[conceptcard]
    {\bfseries\footnotesize\color{gimpanavy} Alphabets, Strings, and Languages (Sipser Ch. 0)}
    \vspace{0.06cm}
    \begin{itemize}\setlength{\itemsep}{2.5pt}\footnotesize
      \item \textbf{Alphabet ($\Sigma$):} A non-empty, finite set of symbols (e.g., binary $\Sigma = \{0, 1\}$). \\
            {\scriptsize\color{gimpaaccent}$\blacktriangleright$ \textbf{Modern AI Connection:} \textbf{LLM Tokenization Alphabets} (e.g., GPT/Llama BPE vocabulary $|\Sigma| \approx 100\text{k}$ token IDs).}
      \item \textbf{String ($w$):} A finite sequence of symbols chosen from $\Sigma$. Length $|w|$ denotes symbol count. The empty string is denoted $\varepsilon$ ($|\varepsilon| = 0$).
      \item \textbf{Language ($L$):} Any set of strings formed over an alphabet ($L \subseteq \Sigma^*$).
      \item \textbf{Kleene Star ($\Sigma^*$):} The set of all possible finite strings over $\Sigma$, including $\varepsilon$: $\Sigma^* = \bigcup_{k=0}^{\infty} \Sigma^k$.
    \end{itemize}
  \end{tcolorbox}

  \vspace{0.08cm}
  \begin{tcolorbox}[examplecard]
    {\bfseries\footnotesize\color{accentGreen} Concrete Language Instance}
    \vspace{0.04cm}
    \footnotesize
    Let $\Sigma = \{0, 1\}$. The language of all binary strings with an even number of zeros is defined as:
    $$L = \bigl\{w \in \Sigma^* \;\big|\; |w|_0 \equiv 0 \pmod 2\bigr\} = \{\varepsilon, 1, 11, 00, 010, 1001, \ldots\}.$$
  \end{tcolorbox}
\end{frame}

\begin{frame}{Week 2: Deterministic Finite Automata (DFA)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \colheader{Formal 5-Tuple Definition (Sipser Def 1.5)}
      \begin{tcolorbox}[conceptcard]
        \footnotesize
        A \textbf{DFA} is a 5-tuple $\bigl(Q, \Sigma, \delta, q_0, F\bigr)$:
        \begin{enumerate}\setlength{\itemsep}{2pt}
          \item $Q$: Finite set of internal \textbf{states}.
          \item $\Sigma$: Finite input \textbf{alphabet}.
          \item $\delta: Q \times \Sigma \to Q$: \textbf{Transition function}.
          \item $q_0 \in Q$: Designated \textbf{start state}.
          \item $F \subseteq Q$: Set of \textbf{accept states}.
        \end{enumerate}
      \end{tcolorbox}
      \vspace{0.05cm}
      \begin{tcolorbox}[examplecard]
        \scriptsize
        \textbf{Determinism Invariant:} For every pair $(q, a) \in Q \times \Sigma$, exactly one transition exists.
      \end{tcolorbox}
    \end{column}

    \hfill

    \begin{column}{0.48\textwidth}
      \colheader{DFA State Diagram: Strings Ending in `1'}
      \vspace{0.05cm}
      \begin{center}
        \begin{tikzpicture}[node distance=2.2cm, auto]
          \node[state, initial] (q0) {$q_0$}; 
          \node[state, accepting] (q1) [right=of q0] {$q_1$}; 
          \path[->] 
          (q0) edge [loop above] node {0} ()
               edge [bend left=25] node {1} (q1)
          (q1) edge [loop above] node {1} ()
               edge [bend left=25] node {0} (q0);
        \end{tikzpicture}
      \end{center}
      \vspace{-0.15cm}
      \begin{tcolorbox}[conceptcard]
        \scriptsize
        \textbf{Trace on $w = 0101$:} \\
        $q_0 \xrightarrow{0} q_0 \xrightarrow{1} q_1 \xrightarrow{0} q_0 \xrightarrow{1} q_1 \in F$ (\textbf{Accepted}).\\[0.05cm]
        \textbf{Complexity:} Strictly $\mathcal{O}(|w|)$ linear time, $\mathcal{O}(1)$ space.
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Week 3: Nondeterministic Finite Automata (NFA)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \colheader{Formal Definition (Sipser Def 1.16)}
      \begin{tcolorbox}[conceptcard]
        \footnotesize
        An \textbf{NFA} is a 5-tuple $\bigl(Q, \Sigma, \delta, q_0, F\bigr)$ where:
        $$\delta: Q \times \Sigma_{\varepsilon} \to \mathcal{P}(Q), \quad \Sigma_{\varepsilon} = \Sigma \cup \{\varepsilon\}$$
        \vspace{-0.15cm}
        \begin{itemize}\setlength{\itemsep}{2pt}\scriptsize
          \item \textbf{Branching:} Transitions to subsets of states.
          \item \textbf{$\varepsilon$-Transitions:} State jumps without input.
          \item \textbf{Acceptance:} Accepted if \textit{at least one} path terminates in an accept state $q \in F$.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    \hfill

    \begin{column}{0.48\textwidth}
      \colheader{Fundamental Equivalence Theorem}
      \begin{tcolorbox}[examplecard]
        \footnotesize
        \textbf{Theorem (Sipser Thm 1.39):} \\
        \textit{Every NFA has an equivalent DFA recognizing the exact same formal language.}
        $$\mathcal{L}(\text{NFA}) = \mathcal{L}(\text{DFA}) = \text{Regular Languages}$$
        \vspace{-0.15cm}
        \begin{itemize}\setlength{\itemsep}{2pt}\scriptsize
          \item \textbf{Proof Method:} Powerset (Subset) Construction.
          \item \textbf{Expressive Equivalence:} Zero added power, but exponential descriptive conciseness ($|Q_{\text{NFA}}| \ll |Q_{\text{DFA}}|$).
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
\begin{frame}{Week 3: The Powerset (Subset) Construction Algorithm}
  \begin{tcolorbox}[conceptcard]
    {\bfseries\footnotesize\color{gimpanavy} Algorithm: Convert NFA $N=(Q, \Sigma, \delta, q_0, F)$ to Equivalent DFA $M=(Q', \Sigma, \delta', q_0', F')$}
    \vspace{0.06cm}
    \begin{enumerate}\setlength{\itemsep}{2pt}\footnotesize
      \item \textbf{State Space Construction:} The DFA state set is the power set of NFA states:
        $$Q' = \mathcal{P}(Q) \implies |Q'| \le 2^{|Q|}.$$
      \item \textbf{DFA Start State:} Include the initial state and all states reachable via $\varepsilon$-transitions:
        $$q_0' = E(\{q_0\}), \quad \text{where } E(R) \text{ is the } \varepsilon\text{-closure of state set } R.$$
      \item \textbf{DFA Transition Function:} For each subset $R \in Q'$ and symbol $a \in \Sigma$:
        $$\delta'(R, a) = \bigcup_{r \in R} E\bigl(\delta(r, a)\bigr).$$
      \item \textbf{DFA Accept States:} Any subset containing at least one original accept state:
        $$F' = \bigl\{R \in Q' \;\big|\; R \cap F \neq \varnothing\bigr\}.$$
    \end{enumerate}
  \end{tcolorbox}

  \vspace{0.08cm}
  \begin{center}
    {\footnotesize\color{gimpaslate}\textbf{Takeaway:} Nondeterminism provides exponential descriptive conciseness without increasing computational power.}
  \end{center}
\end{frame}

\begin{frame}{Week 4: Regular Expressions \& Kleene's Theorem}
  \begin{tcolorbox}[conceptcard]
    {\bfseries\footnotesize\color{gimpanavy} Inductive Definition of Regular Expressions (Sipser Def 1.52)}
    \vspace{0.06cm}
    \footnotesize
    A string $R$ is a regular expression over $\Sigma$ if $R$ is:
    \begin{enumerate}\setlength{\itemsep}{1.5pt}
      \item \textbf{Base Cases:} $a \in \Sigma$ ($\{a\}$), $\varepsilon$ ($\{\varepsilon\}$), or $\varnothing$ ($\emptyset$).
      \item \textbf{Inductive Operations:} Given regular expressions $R_1, R_2$:
        \begin{itemize}\setlength{\itemsep}{1pt}\footnotesize
          \item \textbf{Union:} $R_1 \cup R_2$ (or $R_1 \mid R_2$), recognizing $L(R_1) \cup L(R_2)$.
          \item \textbf{Concatenation:} $R_1 \circ R_2$ (or $R_1 R_2$), recognizing $\{uv \mid u \in L(R_1), v \in L(R_2)\}$.
          \item \textbf{Star (Kleene Closure):} $R_1^*$, recognizing $\bigcup_{k=0}^{\infty} L(R_1)^k$.
        \end{itemize}
    \end{enumerate}
  \end{tcolorbox}

  \vspace{0.08cm}
  \begin{tcolorbox}[examplecard]
    {\bfseries\footnotesize\color{accentGreen} Kleene's Theorem (1956)}
    \vspace{0.04cm}
    \footnotesize
    $$\text{Language } L \text{ is Regular} \iff L \text{ is described by a Regex} \iff L \text{ is recognized by a DFA / NFA.}$$
    \vspace{-0.05cm}
    \textbf{Equivalence Pipeline:} Regex $\xrightarrow{\text{Thompson's Alg.}}$ NFA $\xrightarrow{\text{Subset Const.}}$ DFA $\xrightarrow{\text{State Elimination}}$ Regex.
  \end{tcolorbox}
\end{frame}

\begin{frame}{Week 4 Spotlight: Catastrophic ReDoS Vulnerabilities}
  \begin{tcolorbox}[spotlightcard]
    \footnotesize
    \textbf{Seminal Literature (USENIX Security 2018):} \\
    \textbf{Staicu \& Pradel (TU Darmstadt)}, \textit{``Freezing the Web: A Study of ReDoS Vulnerabilities in JavaScript-based Web Servers,''} \textbf{USENIX Security Symposium 2018}.
  \end{tcolorbox}

  \vspace{0.06cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \colheader{Vulnerability: Backtracking Engines}
      \begin{tcolorbox}[alertcard]
        \begin{itemize}\setlength{\itemsep}{2.5pt}\scriptsize
          \item \textbf{Engine Flaw:} Python \texttt{re}, Node.js (V8), and Java implement regex via recursive depth-first backtracking.
          \item \textbf{Pathological Regex:} A nested quantifier like \texttt{(a+)+b} on input string $a^n$ requires $\mathcal{O}(2^n)$ evaluations.
          \item \textbf{Denial of Service (ReDoS):} A 35-character malicious HTTP payload freezes Node.js event loops for 10+ hours ($100\%$ CPU spike).
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    \hfill

    \begin{column}{0.48\textwidth}
      \colheader{Solution: Automata-Theoretic Execution}
      \begin{tcolorbox}[conceptcard]
        \begin{itemize}\setlength{\itemsep}{2.5pt}\scriptsize
          \item \textbf{The Thompson NFA Remedy:} Modern secure engines (Google \textbf{RE2}, Rust \texttt{regex}) abandon backtracking.
          \item \textbf{Simultaneous Set Tracking:} The engine tracks all active NFA states in lockstep:
            $$\mathcal{O}(|w| \cdot |Q|) \text{ strictly linear worst-case time!}$$
          \item \textbf{Immunity:} Completely immune to algorithmic complexity attacks by theoretical construction.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Week 5: Limits of Finite Automata \& Pigeonhole Principle}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.49\textwidth}
      \colheader{The Pigeonhole Principle \& Loops}
      \begin{tcolorbox}[conceptcard]
        \begin{itemize}\setlength{\itemsep}{2.5pt}\footnotesize
          \item Let $M$ be a DFA with $p$ states ($|Q| = p$).
          \item If $M$ processes any string $s \in L(M)$ with length $|s| \ge p$, the path visits at least $p + 1$ states.
          \item By the \textbf{Pigeonhole Principle}, at least one state $q_j \in Q$ must be visited \textbf{twice}.
          \item The computation necessarily contains a \textbf{directed cycle (loop)}.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    \hfill

    \begin{column}{0.48\textwidth}
      \colheader{Automata Cycle Decomposition}
      \vspace{0.05cm}
      \begin{center}
        \begin{tikzpicture}[node distance=1.35cm, auto]
          \node[state, initial] (q0) {$q_0$}; 
          \node[state] (q1) [right=of q0] {$q_j$}; 
          \node[state, accepting] (q2) [right=of q1] {$q_{\text{acc}}$}; 
          \path[->] 
          (q0) edge node {$x$} (q1)
          (q1) edge [loop above] node {$y$ (loop)} (q1)
               edge node {$z$} (q2);
        \end{tikzpicture}
      \end{center}
      \vspace{-0.15cm}
      \footnotesize
      Decompose string $s = xyz$ where:
      \begin{itemize}\setlength{\itemsep}{1.5pt}\scriptsize
        \item $x$: prefix driving $q_0 \to q_j$.
        \item $y$: loop cycle driving $q_j \to q_j$ ($|y| > 0$).
        \item $z$: suffix driving $q_j \to q_{\text{acc}}$.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.06cm}
  \begin{center}
    \parbox{0.94\textwidth}{%
      \centering\footnotesize\color{gimpanavy}
      \textbf{Core Consequence:} The cycle $y$ can be pumped $i = 0, 1, 2, \dots$ times; the automaton cannot prevent $x y^i z$ from being accepted, proving language regularity.
    }
  \end{center}
\end{frame}

\begin{frame}{Week 5: The Pumping Lemma for Regular Languages}
  \begin{tcolorbox}[conceptcard]
    {\bfseries\footnotesize\color{gimpanavy} Formal Theorem Statement (Sipser Thm 1.70)}
    \vspace{0.04cm}
    \footnotesize
    If $L$ is a regular language, then there exists a pumping length $p \ge 1$ such that every string $s \in L$ with $|s| \ge p$ can be divided into $s = xyz$ satisfying three conditions:
    \begin{enumerate}\setlength{\itemsep}{1pt}
      \item $x y^i z \in L$ for each $i \ge 0$ \quad (\textbf{Pumping Property}),
      \item $|y| > 0$ \quad (\textbf{Non-empty Loop}),
      \item $|xy| \le p$ \quad (\textbf{Loop Occurs within First $p$ Characters}).
    \end{enumerate}
  \end{tcolorbox}

  \vspace{0.06cm}
  \begin{tcolorbox}[alertcard]
    {\bfseries\footnotesize\color{accentCrimson} Adversarial Proof Template: Showing $L = \{0^n 1^n \mid n \ge 0\}$ is Non-Regular}
    \vspace{0.04cm}
    \begin{enumerate}\setlength{\itemsep}{1.5pt}\scriptsize
      \item \textbf{Proof by Contradiction:} Assume $L$ is regular. The Pumping Lemma guarantees a pumping length $p$.
      \item \textbf{Select String:} Choose $s = 0^p 1^p \in L$. Observe that $|s| = 2p \ge p$.
      \item \textbf{Constrain Substring $y$:} Since $|xy| \le p$, substring $y$ lies entirely within the initial block of zeros: $y = 0^k$ for $k \ge 1$.
      \item \textbf{Pump Down ($i = 0$):} Consider $s' = x y^0 z = x z$. Symbol count: $|s'|_0 = p - k \neq p = |s'|_1$.
      \item \textbf{Contradiction:} $s' \notin L$, violating Condition 1 $\implies$ \textbf{$L$ is strictly non-regular!}
    \end{enumerate}
  \end{tcolorbox}
\end{frame}

\section*{Problem 1: Undecidability via Mapping Reductions (25 Marks)}
Let $E_{\text{TM}} = \{\langle M \rangle \mid M \text{ is a TM and } L(M) = \varnothing\}$.
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Prove that $E_{\text{TM}}$ is **undecidable** by showing a mapping reduction $A_{\text{TM}} \le_{\text{m}} \overline{E_{\text{TM}}}$ (or reduction from $A_{\text{TM}}$ to $E_{\text{TM}}$) (Sipser Theorem 5.2). (15 Marks)
    \item Use your result to prove that $EQ_{\text{TM}} = \{\langle M_1, M_2 \rangle \mid L(M_1) = L(M_2)\}$ is also undecidable. (10 Marks)
\end{enumerate}

\vspace{8pt}

\section*{Problem 2: Rice's Theorem Applications (25 Marks)}
\textbf{Rice's Theorem} states that any non-trivial semantic property of a Turing machine's language is undecidable.
Using Rice's Theorem, determine (with rigorous justification) whether each of the following languages is decidable or undecidable:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $L_1 = \{\langle M \rangle \mid M \text{ is a TM and } L(M) \text{ contains the string } \texttt{101}\}$. (8 Marks)
    \item $L_2 = \{\langle M \rangle \mid M \text{ is a TM and } L(M) \text{ is a context-free language}\}$. (8 Marks)
    \item $L_3 = \{\langle M \rangle \mid M \text{ is a TM and } M \text{ has at most 5 states}\}$. (9 Marks)
\end{enumerate}

\vspace{8pt}

\section*{Problem 3: NP-Completeness Reduction: 3SAT to CLIQUE (25 Marks)}
A **clique** in an undirected graph $G = (V, E)$ is a subset of vertices $V' \subseteq V$ such that every pair of distinct vertices in $V'$ is connected by an edge. Let:
$$\text{CLIQUE} = \{\langle G, k \rangle \mid G \text{ is an undirected graph containing a clique of size } k\}$$
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Prove that $\text{CLIQUE} \in \mathbf{NP}$ by defining a polynomial-time verification algorithm. (8 Marks)
    \item Given the 3CNF Boolean formula $\phi = (x_1 \lor \overline{x_2} \lor x_3) \land (\overline{x_1} \lor x_2 \lor x_3) \land (x_1 \lor x_2 \lor \overline{x_3})$, construct the exact graph $G_\phi$ and target integer $k$ according to the Cook-Karp 3SAT-to-CLIQUE reduction (Sipser Theorem 7.32). Draw the graph $G_\phi$. (10 Marks)
    \item Prove that $\phi$ is satisfiable $\iff G_\phi$ contains a clique of size $k=3$. (7 Marks)
\end{enumerate}

\vspace{8pt}

\section*{Problem 4: Modern Applied Case Study --- Ethereum Smart Contract Gas (25 Marks)}
The Ethereum Virtual Machine (EVM) executes Turing-complete bytecode submitted by untrusted network users.
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Prove why it is mathematically impossible for an Ethereum miner to statically pre-screen smart contracts to reject all transactions containing infinite loops before execution begins. (12 Marks)
    \item Formulate how Ethereum's deterministic Gas metering mechanism solves the practical consequences of Turing's Halting Problem, guaranteeing bounded execution time for network consensus. (13 Marks)
\end{enumerate}
